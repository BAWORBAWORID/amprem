Otentikasi API

Semua request API wajib menyertakan autentikasi berbasis API Key. Jika key tidak valid atau masa aktif paket API Anda habis, server akan mengembalikan status error 401 atau 403.

Contoh Script Bot

Berikut adalah contoh script integrasi untuk bot WhatsApp/Telegram Anda yang menggunakan API kami (untuk role Auto Generator atau API Key Premium).

1. settings.js

Konfigurasikan API Key dan domain default Anda pada file settings bot:

// settings.js
module.exports = {
    apiKey: "__API_KEY__",
    apiUrl: "https://am.alwayscodex.eu.cc/api",
    defaultDomain: "softbank.id"
};

2. index.js

Contoh kode NodeJS untuk mengotomatiskan pemanggilan API Generator Otomatis:

// index.js
const axios = require('axios');
const config = require('./settings');

async function generatePremiumAccount() {
    try {
        console.log("[Bot] Memulai batch generator otomatis...");
        
        // 1. Memulai batch 1 akun menggunakan domain pilihan
        const startRes = await axios.post(`${config.apiUrl}/am/autogen/start-batch`, {
            domain: config.defaultDomain,
            count: 1
        }, {
            headers: { 'x-api-key': config.apiKey }
        });

        if (!startRes.data.success) {
            throw new Error(startRes.data.error || "Gagal memulai batch.");
        }

        console.log("[Bot] Batch berhasil dimulai di server. Mulai memantau progres...");

        // 2. Lakukan polling status setiap 3 detik hingga selesai
        let completed = false;
        while (!completed) {
            await new Promise(resolve => setTimeout(resolve, 3000));

            const statusRes = await axios.get(`${config.apiUrl}/am/autogen/active-batch`, {
                headers: { 'x-api-key': config.apiKey }
            });

            const job = statusRes.data.job;
            if (!job) {
                console.log("[Bot] Tidak ada batch aktif.");
                break;
            }

            console.log(`[Bot] Status: ${job.status} | Berhasil: ${job.successfulCount}/${job.totalCount}`);

            if (job.status === 'completed' || job.status === 'failed') {
                completed = true;
                if (job.results && job.results.length > 0) {
                    const account = job.results[0];
                    console.log(`\n[Bot] GENERATE SUKSES!`);
                    console.log(`Email: ${account.email}`);
                    console.log(`Inbox: ${account.inbox}\n`);
                } else {
                    console.log(`[Bot] Pembuatan akun gagal.`);
                }
            }
        }
    } catch (error) {
        console.error("[Bot] Eror:", error.response ? error.response.data : error.message);
    }
}

generatePremiumAccount();

const axios = require('axios');

const API_KEY = '__API_KEY__';
const BASE_URL = 'https://am.alwayscodex.eu.cc/api/v1/bot-premium';
const headers = { 'x-api-key': API_KEY };

async function callApi() {
  try {
    const sendLink = await axios.post(
      `${BASE_URL}/send-link`,
      { email: 'target@email.com' },
      { headers }
    );
    console.log('Send Link:', sendLink.data);

    const activate = await axios.post(
      `${BASE_URL}/activate`,
      {
        email: 'target@email.com',
        magicLink: 'https://alightcreative.com/...'
      },
      { headers }
    );
    console.log('Activate:', activate.data);
  } catch (error) {
    console.error('API Error:', error.response?.data || error.message);
  }
}

import requests

API_KEY = '__API_KEY__'

try:

    activate = requests.post(
        f'{BASE_URL}/activate',
        json={
            'email': 'target@email.com',
            'magicLink': 'https://alightcreative.com/...',
        },
        headers=HEADERS,
        timeout=30,
    )
    activate.raise_for_status()
    print('Activate:', activate.json())
except requests.RequestException as error:
    print('API Error:', error)
